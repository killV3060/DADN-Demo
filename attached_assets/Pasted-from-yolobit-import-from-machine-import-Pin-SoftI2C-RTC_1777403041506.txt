from yolobit import *
from machine import Pin, SoftI2C, RTC
from aiot_dht20 import DHT20
from aiot_lcd1602 import LCD1602
from aiot_rgbled import RGBLed
from wifi import __wifi__
import ntptime
import time
import utime
import ujson

# Thư viện chứa class MQTTClient (khởi tạo và xử lý MQTT)
from umqtt.simple import MQTTClient

# ================= CONFIG =================
WIFI_SSID = "Vinh"
WIFI_PASS = "12345678"

MQTT_BROKER = "34.1.136.26"
MQTT_PORT = 1883
DEVICE_ID = "1"

TOPIC_CMD = b"yolobit/command/" + DEVICE_ID.encode()
TOPIC_DATA = b"yolobit/sensor/" + DEVICE_ID.encode()

# ================= GLOBAL =================
client = None
last_publish = 0
PUBLISH_INTERVAL = 10000  # 10s

Temperature = 0
Humidity = 0
Luminosity = 0

temp_max = 35
humid_min = 40

# ================= HARDWARE =================
dht = DHT20()
lcd = LCD1602()
rgb = RGBLed(pin1.pin, 4)

# ================= WIFI =================
def connect_wifi():
  print("Connecting WiFi...")
  __wifi__.connect_wifi(WIFI_SSID, WIFI_PASS)
  while not __wifi__.isconnected():
    time.sleep(1)
  print("WiFi connected")

# ================= MQTT =================
def mqtt_callback(topic, msg):
  global temp_max, humid_min
  cmd = None

  # Dashboard/API gửi payload JSON: {"command":"5", ...}
  try:
    payload = ujson.loads(msg.decode())
    if isinstance(payload, dict) and "command" in payload:
      cmd = str(payload["command"])
  except:
    pass

  # Fallback cho trường hợp gửi plain text
  if cmd is None:
    cmd = msg.decode()

  print("CMD:", cmd)

  # ===== COMMAND HANDLER =====
  # Dashboard sẽ gửi các lệnh như sau:

  # "1": Quay servo về 0 độ
  if cmd == "1":
    pin4.servo_write(0, max=270)
  # "2": Quay servo về 180 độ
  elif cmd == "2":
    pin4.servo_write(180, max=270)
  # "3": Bật LED đỏ
  elif cmd == "3":
    rgb.set_all('#ff0000')
  # "4": Tắt LED
  elif cmd == "4":
    rgb.set_all('#000000')

  # Điều khiển quạt
  elif cmd == "FAN:0":
    pin10.write_analog(0)  # tắt quạt
  elif cmd == "FAN:33":
    pin10.write_analog(341) # bật quạt 33%
  elif cmd == "FAN:100":
    pin10.write_analog(1023) # bật quạt max


  # Đặt ngưỡng nhiệt độ tối đa
  elif cmd.startswith("TEMP:"):
    temp_max = int(cmd.split(":")[1])
  # Đặt ngưỡng độ ẩm tối thiểu
  elif cmd.startswith("HUMID:"):
    humid_min = int(cmd.split(":")[1])

def connect_mqtt():
  global client

  while True:
    try:
      print("Connecting MQTT...")
      client = MQTTClient("yolobit_" + DEVICE_ID, MQTT_BROKER, port=MQTT_PORT)
      client.set_callback(mqtt_callback)
      client.connect()
      client.subscribe(TOPIC_CMD)
      print("MQTT connected")
      return
    except Exception as e:
      print("MQTT connect failed:", e)
      time.sleep(2)

# ================= SENSOR =================
def read_sensor():
  global Temperature, Humidity, Luminosity

  dht.read_dht20()
  Temperature = dht.dht20_temperature()
  Humidity = dht.dht20_humidity()
  Luminosity = round(translate(pin0.read_analog(), 0, 4095, 0, 100))

# ================= LCD =================
def update_lcd():
  lcd.move_to(0, 0)
  lcd.putstr(str(Temperature) + "*C ")

  lcd.move_to(8, 0)
  lcd.putstr(str(Humidity) + "% ")

# ================= CONTROL LOGIC =================
def auto_control():
  if Temperature > temp_max:
    pin10.write_analog(512)  # bật quạt
  else:
    pin10.write_analog(0)

  if Humidity < humid_min:
    pin14.write_analog(512)
  else:
    pin14.write_analog(0)

# ================= PUBLISH =================
def publish_data():
  data = {
    "deviceId": DEVICE_ID,
    "source": "mqtt-device",
    "temperature": Temperature,
    "humidity": Humidity,
    "luminosity": Luminosity,
    "timestamp": utime.time()
  }

  try:
    client.publish(TOPIC_DATA, ujson.dumps(data))
    print("Published:", data)
  except:
    print("Publish failed")

# ================= SETUP =================
connect_wifi()

ntptime.settime()
rtc = RTC()
(y, m, d, wd, h, mi, s, ms) = rtc.datetime()
h = (h + 7) % 24  # UTC+7
rtc.init((y, m, d, wd, h, mi, s, ms))

connect_mqtt()

lcd.clear()
lcd.putstr("IoT Ready")

# ================= MAIN LOOP =================
while True:
  try:
    # giữ MQTT alive
    client.check_msg()

    # đọc sensor
    read_sensor()

    # update LCD
    update_lcd()

    # auto control
    auto_control()

    # gửi data mỗi 10s
    now = time.ticks_ms()
    if time.ticks_diff(now, last_publish) > PUBLISH_INTERVAL:
      publish_data()
      last_publish = now

    time.sleep_ms(200)

  except Exception as e:
    print("Error:", e)

    # reconnect MQTT nếu lỗi
    try:
      connect_mqtt()
    except:
      pass

    time.sleep(2)
